cmake_minimum_required(VERSION 3.16)
project(ObjectTreeDemo LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
find_package(QT NAMES Qt6 Qt5 REQUIRED COMPONENTS Widgets)
find_package(Qt${QT_VERSION_MAJOR} REQUIRED COMPONENTS Widgets)
if(MSVC)
    add_compile_options(/utf-8)
endif()
add_executable(ObjectTreeDemo main.cpp mainwindow.cpp mainwindow.h widgets.h)
target_link_libraries(ObjectTreeDemo PRIVATE Qt${QT_VERSION_MAJOR}::Widgets)
include(CTest)
if(BUILD_TESTING)
    find_package(Qt${QT_VERSION_MAJOR} REQUIRED COMPONENTS Test)
    add_executable(ownership_tests tests/ownership_tests.cpp mainwindow.cpp mainwindow.h widgets.h)
    target_include_directories(ownership_tests PRIVATE ${CMAKE_CURRENT_SOURCE_DIR})
    target_link_libraries(ownership_tests PRIVATE Qt${QT_VERSION_MAJOR}::Widgets Qt${QT_VERSION_MAJOR}::Test)
    add_test(NAME ownership_tests COMMAND ownership_tests -platform offscreen)
endif()
